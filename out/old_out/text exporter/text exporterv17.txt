# compile_po.py
import polib
from pathlib import Path

def ensure_locale_structure():
    """Crée les dossiers et fichiers .po s'ils n'existent pas."""
    locale_dir = Path("locale")
    languages = ["fr", "en"]

    for lang in languages:
        po_dir = locale_dir / lang / "LC_MESSAGES"
        po_dir.mkdir(parents=True, exist_ok=True)
        po_file = po_dir / "messages.po"

        if not po_file.exists():
            # Créer un fichier .po minimal avec en-tête
            po_content = f'''msgid ""
msgstr ""
"Project-Id-Version: PythonCodeExtractor 1.0\\n"
"POT-Creation-Date: 2026-08-13 11:00+0200\\n"
"PO-Revision-Date: 2026-08-13 11:00+0200\\n"
"Last-Translator: \\n"
"Language-Team: \\n"
"Language: {lang}\\n"
"MIME-Version: 1.0\\n"
"Content-Type: text/plain; charset=UTF-8\\n"
"Content-Transfer-Encoding: 8bit\\n"
'''
            with open(po_file, 'w', encoding='utf-8') as f:
                f.write(po_content)
            print(f"✅ Fichier {po_file} créé.")

def compile_all_po():
    """Compile tous les fichiers .po en .mo."""
    locale_dir = Path("locale")
    if not locale_dir.exists():
        print("❌ Le dossier 'locale' n'existe pas. Création...")
        ensure_locale_structure()

    for lang_dir in locale_dir.iterdir():
        if lang_dir.is_dir():
            po_file = lang_dir / "LC_MESSAGES" / "messages.po"
            mo_file = lang_dir / "LC_MESSAGES" / "messages.mo"
            if po_file.exists():
                try:
                    po = polib.pofile(str(po_file))
                    po.save_as_mofile(str(mo_file))
                    print(f"✅ {mo_file} généré avec succès")
                except Exception as e:
                    print(f"❌ Erreur pour {po_file} : {e}")
            else:
                print(f"⚠️  {po_file} manquant, ignoré.")

if __name__ == "__main__":
    ensure_locale_structure()
    compile_all_po()
    print("✅ Compilation terminée.")

# src/config.py
import json
from pathlib import Path

CONFIG_PATH = Path(__file__).parent.parent / "config.json"

class Config:
    """Charge et fournit l'accès à la configuration centralisée."""

    _instance = None
    _config = {}

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._load_config()
        return cls._instance

    @classmethod
    def _load_config(cls):
        """Charge le fichier config.json, ou utilise des valeurs par défaut."""
        default_config = {
            "output_dir": "out",
            "received_subdir": "received",
            "archive_subdir": "old_out",
            "version_file": "extractor_version.txt",
            "network": {
                "server_host": "0.0.0.0",
                "server_port": 50000,
                "discovery_port": 50001,
                "broadcast_msg": "PYEXTRACTOR_DISCOVER",
                "reply_msg": "PYEXTRACTOR_HERE"
            },
            "extraction": {
                "include_json": True,
                "include_subdirs": True,
                "show_file_paths": True,
                "include_structure": True,
                "include_txt": False,
                "ignore_init": False,
                "include_statistics": True,
                "include_file_metadata": False
            },
            "gui": {
                "window_width": 700,
                "window_height": 600,
                "log_height": 12
            }
        }

        if CONFIG_PATH.exists():
            try:
                with open(CONFIG_PATH, 'r', encoding='utf-8') as f:
                    user_config = json.load(f)
                # Mise à jour récursive (fusion simple)
                cls._config = cls._merge(default_config, user_config)
            except Exception as e:
                print(f"Erreur de chargement de la configuration : {e} – utilisation des valeurs par défaut")
                cls._config = default_config
        else:
            cls._config = default_config

    @staticmethod
    def _merge(base, override):
        """Fusion récursive de deux dictionnaires."""
        result = base.copy()
        for key, value in override.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = Config._merge(result[key], value)
            else:
                result[key] = value
        return result

    def get(self, key, default=None):
        """Récupère une valeur de configuration via une clé pointée (ex: 'network.server_port')."""
        keys = key.split('.')
        value = self._config
        for k in keys:
            if isinstance(value, dict) and k in value:
                value = value[k]
            else:
                return default
        return value

    def get_all(self):
        """Retourne tout le dictionnaire de configuration."""
        return self._config

# Instance unique
config = Config()

# src/logger.py
import logging
import sys

def setup_logger(name=None, level=logging.INFO):
    """Configure et retourne un logger avec un format standard."""
    logger = logging.getLogger(name or __name__)
    if not logger.handlers:
        handler = logging.StreamHandler(sys.stdout)
        handler.setFormatter(logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s',
            datefmt='%Y-%m-%d %H:%M:%S'
        ))
        logger.addHandler(handler)
        logger.setLevel(level)
    return logger

# versioning.py
from pathlib import Path
from src.logger import setup_logger

logger = setup_logger(__name__)

class VersionManager:
    def __init__(self, version_file="extractor_version.txt"):
        self.version_file = Path(version_file)
        self.mapping = self.load_mapping()

    def load_mapping(self):
        mapping = {}
        if self.version_file.exists():
            try:
                with open(self.version_file, "r", encoding="utf-8") as f:
                    for line in f:
                        line = line.strip()
                        if line and ":" in line:
                            folder, ver = line.split(":", 1)
                            mapping[folder] = int(ver)
            except (IOError, OSError, ValueError) as e:
                logger.error(f"Erreur lors du chargement du fichier de versions : {e}")
                mapping = {}
        return mapping

    def save_mapping(self):
        try:
            with open(self.version_file, "w", encoding="utf-8") as f:
                for folder, ver in sorted(self.mapping.items()):
                    f.write(f"{folder}:{ver}\n")
        except (IOError, OSError) as e:
            logger.error(f"Erreur lors de la sauvegarde du fichier de versions : {e}")

    def get_next_version(self, folder_name, output_dir="."):
        output_dir = Path(output_dir)
        last_version = self.mapping.get(folder_name, 0)
        version = last_version + 1
        while (output_dir / f"{folder_name}v{version}.txt").exists():
            version += 1
        return version

    def use_version(self, folder_name, version):
        self.mapping[folder_name] = version
        self.save_mapping()

    def reset(self):
        """Réinitialise complètement l'historique des versions."""
        self.mapping = {}
        if self.version_file.exists():
            try:
                self.version_file.unlink()
                logger.info("Fichier de versions supprimé.")
            except (IOError, OSError) as e:
                logger.error(f"Erreur lors de la suppression du fichier de versions : {e}")

    # Anciennes méthodes, conservées pour compatibilité
    def load_version(self):
        return 1

    def save_version(self, version):
        pass

# src/extractor/content_formatter.py
import json

def format_json_content(content, file_path=None):
    """Essaie de formater joliment le contenu JSON.
       En cas d'échec, retourne le contenu brut précédé d'un commentaire d'erreur.
    """
    try:
        json_data = json.loads(content)
        return json.dumps(json_data, indent=2, ensure_ascii=False)
    except json.JSONDecodeError as e:

    @staticmethod
    def _human_size(size_bytes):
        if size_bytes < 1024:
            return f"{size_bytes} octets"
        elif size_bytes < 1024 * 1024:
            return f"{size_bytes / 1024:.2f} Ko"
        else:
            return f"{size_bytes / (1024 * 1024):.2f} Mo"

    def _write_header(self, output_file, folder):
        from src.utils import get_current_date
        output_file.write(f"Extraction du code du dossier : {folder}\n")
        output_file.write(f"Date d'extraction : {get_current_date()}\n")
        output_file.write("=" * 80 + "\n\n")

    def _process_file(self, full_path, ext):
        full_path = Path(full_path)
        try:
            # Pour les fichiers .mo, lecture binaire et encodage base64
            if ext == '.mo':
                with open(full_path, 'rb') as f:
                    raw = f.read()
                content = base64.b64encode(raw).decode('ascii')
                num_lines = len(content.splitlines())
                file_size = len(raw)
                read_ok = True
            else:
                with open(full_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                if ext == '.json':
                    content = self.format_json_content(content, full_path)
                num_lines = len(content.splitlines())
                file_size = full_path.stat().st_size
                read_ok = True
        except Exception as e:
            logger.error(f"Erreur de lecture du fichier {full_path} : {e}")
            content = f"// ERREUR: Impossible de lire le fichier : {str(e)}\n"
            num_lines = 0
            file_size = 0
            read_ok = False
        return content, num_lines, file_size, read_ok

    def _write_file_section(self, output_file, full_path, rel_path, ext, file_type,
                            content, num_lines, file_size, read_ok):
        full_path = Path(full_path)
        rel_path = Path(rel_path)
        output_file.write(f"\n{'=' * 80}\n")
        output_file.write(f"FICHIER {file_type} : ")

        else:
            output_file.write(f"{full_path.name}\n")
        output_file.write(f"Type: {file_type}\n")
        output_file.write(f"Chemin complet: {full_path}\n")

        if self.include_file_metadata:
            parent_dir = rel_path.parent if self.show_file_paths else full_path.parent
            if not parent_dir or str(parent_dir) == '.':
                parent_dir = "."
            output_file.write(f"Dossier parent: {parent_dir}\n")
            output_file.write(f"Taille: {self._human_size(file_size)}\n")
            output_file.write(f"Nombre de lignes: {num_lines}\n")

        output_file.write("-" * 80 + "\n")
        # Si c'est un fichier .mo, on ajoute un commentaire indiquant le format
        if ext == '.mo' and read_ok:
            output_file.write("// Fichier binaire (.mo) encodé en base64\n")
            output_file.write(content)
            if not content.endswith('\n'):
                output_file.write("\n")
        else:
            output_file.write(content)
            if not content.endswith('\n'):
                output_file.write("\n")

        total_lines_json = 0
        total_lines_txt = 0
        total_lines_po = 0
        total_lines_mo = 0

        output_path = Path(output_filename)
        output_path.parent.mkdir(parents=True, exist_ok=True)

        try:

# src/extractor/file_finder.py
from pathlib import Path

def find_files(folder, include_json=True, include_subdirs=True,
               include_txt=False, include_po=False, include_mo=False, ignore_init=False):
    """
    Trouve tous les fichiers .py, .json, .txt, .po et .mo dans le dossier.
    Retourne une liste de tuples (chemin_complet, chemin_relatif, extension).
    Les chemins sont des objets Path.
    """
    folder = Path(folder)
    files = []
    extensions = ['.py']
    if include_json:
        extensions.append('.json')
    if include_txt:
        extensions.append('.txt')
    if include_po:
        extensions.append('.po')
    if include_mo:
        extensions.append('.mo')

    if include_subdirs:
        for file_path in folder.rglob('*'):
            if file_path.is_file() and file_path.suffix in extensions:
                if ignore_init and file_path.name == '__init__.py':
                    continue
                rel_path = file_path.relative_to(folder)
                files.append((file_path, rel_path, file_path.suffix))
    else:
        for file_path in folder.iterdir():
            if file_path.is_file() and file_path.suffix in extensions:
                if ignore_init and file_path.name == '__init__.py':
                    continue
                files.append((file_path, Path(file_path.name), file_path.suffix))
    return files

from pathlib import Path

def generate_project_structure(folder, include_json=True, include_subdirs=True,
                               include_txt=False, include_po=False, include_mo=False,
                               ignore_init=False):
    folder = Path(folder)
    lines = []
    lines.append("STRUCTURE DU PROJET")
    lines.append("=" * 80)

    root_name = folder.name
    lines.append(f"📁 {root_name}/")

    if include_subdirs:
        for root in sorted(folder.rglob('*')):
            if not root.is_dir():
                continue
            level = len(root.relative_to(folder).parts) - 1
            indent = "│   " * level if level > 0 else ""
            if root != folder:
                lines.append(f"{indent}├── 📁 {root.name}/")

            file_indent = "│   " * (level + 1) if root != folder else ""
            py_files = []
            json_files = []
            txt_files = []
            po_files = []
            mo_files = []
            for f in root.iterdir():
                if f.is_file():
                    if f.suffix == '.py' and not (ignore_init and f.name == '__init__.py'):
                        py_files.append(f.name)
                    elif f.suffix == '.json' and include_json:
                        json_files.append(f.name)
                    elif f.suffix == '.txt' and include_txt:
                        txt_files.append(f.name)
                    elif f.suffix == '.po' and include_po:
                        po_files.append(f.name)
                    elif f.suffix == '.mo' and include_mo:
                        mo_files.append(f.name)
            all_files = sorted(py_files + json_files + txt_files + po_files + mo_files)
            for i, file in enumerate(all_files):
                is_last = (i == len(all_files) - 1)
                prefix = "└── " if is_last else "├── "
                if file in py_files:
                    icon = "🐍"
                elif file in json_files:
                    icon = "📄"
                elif file in po_files:
                    icon = "🌐"
                elif file in mo_files:
                    icon = "📦"
                else:
                    icon = "📝"
                lines.append(f"{file_indent}{prefix} {icon} {file}")
    else:
        py_files = []
        json_files = []
        txt_files = []
        po_files = []
        mo_files = []
        for f in folder.iterdir():
            if f.is_file():
                if f.suffix == '.py' and not (ignore_init and f.name == '__init__.py'):
                    py_files.append(f.name)
                elif f.suffix == '.json' and include_json:
                    json_files.append(f.name)
                elif f.suffix == '.txt' and include_txt:
                    txt_files.append(f.name)
                elif f.suffix == '.po' and include_po:
                    po_files.append(f.name)
                elif f.suffix == '.mo' and include_mo:
                    mo_files.append(f.name)
        all_files = sorted(py_files + json_files + txt_files + po_files + mo_files)
        for i, file in enumerate(all_files):
            is_last = (i == len(all_files) - 1)
            prefix = "└── " if is_last else "├── "
            if file in py_files:
                icon = "🐍"
            elif file in json_files:
                icon = "📄"
            elif file in po_files:
                icon = "🌐"
            elif file in mo_files:
                icon = "📦"
            else:
                icon = "📝"
            lines.append(f"    {prefix} {icon} {file}")

# src/network/discovery.py
import socket
import threading
import time
from src.config import config

class DiscoveryService:
    """Découvre les autres instances du programme sur le réseau local."""

    def __init__(self, listen_port=None):
        self.listen_port = listen_port or config.get('network.discovery_port', 50001)
        self._peers = {}
        self._lock = threading.Lock()
        self.broadcast_msg = config.get('network.broadcast_msg', 'PYEXTRACTOR_DISCOVER').encode()
        self.reply_msg = config.get('network.reply_msg', 'PYEXTRACTOR_HERE').encode()

    def start_listener(self):
        threading.Thread(target=self._listen, daemon=True).start()

    def _listen(self):
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP) as sock:
            sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
            sock.bind(('', self.listen_port))
            while True:
                data, addr = sock.recvfrom(1024)
                if data == self.broadcast_msg:
                    hostname = socket.gethostname()
                    reply = f"{self.reply_msg.decode()}:{hostname}".encode()
                    sock.sendto(reply, addr)
                elif data.startswith(self.reply_msg):
                    parts = data.decode().split(':', 1)
                    if len(parts) == 2:
                        hostname = parts[1]
                        with self._lock:
                            self._peers[addr[0]] = hostname

    def scan(self, timeout=2):
        self._peers.clear()
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP) as sock:
            sock.setsockopt(socket.SOL_SOCKET, socket.SO_BROADCAST, 1)
            sock.sendto(self.broadcast_msg, ('<broadcast>', self.listen_port))
        time.sleep(timeout)
        with self._lock:
            return dict(self._peers)

# src/network/server.py
import socket
import threading
from pathlib import Path
from datetime import datetime
from src.logger import setup_logger
from src.config import config

logger = setup_logger(__name__)

class ReceiveServer(threading.Thread):
    """Serveur TCP qui écoute les transferts de fichiers entrants."""

    def __init__(self, host=None, port=None, received_dir=None):
        super().__init__(daemon=True)
        self.host = host or config.get('network.server_host', '0.0.0.0')
        self.port = port or config.get('network.server_port', 50000)

        # Construction correcte du chemin avec Path
        if received_dir is None:
            output_dir = Path(config.get('output_dir', 'out'))
            received_subdir = config.get('received_subdir', 'received')
            self.received_dir = output_dir / received_subdir
        else:
            self.received_dir = Path(received_dir)

        self._running = True

    def run(self):
        self.received_dir.mkdir(parents=True, exist_ok=True)
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server_socket:
            server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
            try:
                server_socket.bind((self.host, self.port))
                server_socket.listen(1)
                server_socket.settimeout(1.0)
                logger.info(f"Serveur démarré sur {self.host}:{self.port}")
            except Exception as e:
                logger.error(f"Impossible de démarrer le serveur : {e}")
                return

            while self._running:
                try:
                    conn, addr = server_socket.accept()
                except socket.timeout:
                    continue
                except Exception as e:
                    logger.error(f"Erreur lors de l'acceptation : {e}")
                    break
                self._handle_client(conn, addr)

    def _handle_client(self, conn, addr):
        try:
            name_size_bytes = conn.recv(4)
            if not name_size_bytes:
                logger.warning(f"Connexion fermée par {addr} avant l'envoi du nom")
                return
            name_size = int.from_bytes(name_size_bytes, 'big')
            filename = conn.recv(name_size).decode('utf-8')
            data_size_bytes = conn.recv(8)
            if not data_size_bytes:
                logger.warning(f"Connexion fermée par {addr} avant l'envoi de la taille")
                return
            data_size = int.from_bytes(data_size_bytes, 'big')
            data = b''
            while len(data) < data_size:
                chunk = conn.recv(min(4096, data_size - len(data)))
                if not chunk:
                    logger.warning(f"Connexion interrompue par {addr} pendant le transfert")
                    break
                data += chunk

            if len(data) != data_size:
                logger.error(f"Taille reçue incorrecte pour {filename} de {addr} : attendu {data_size}, reçu {len(data)}")
                return

            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            safe_name = f"{timestamp}_{filename}"
            filepath = self.received_dir / safe_name
            with open(filepath, 'wb') as f:
                f.write(data)
            logger.info(f"Fichier reçu de {addr}: {filepath} ({data_size} octets)")
        except socket.error as e:
            logger.error(f"Erreur socket avec {addr} : {e}")
        except Exception as e:
            logger.error(f"Erreur inattendue lors de la réception de {addr} : {e}")
        finally:
            conn.close()

    def stop(self):
        self._running = False
        logger.info("Serveur arrêté")

        self.version_manager = VersionManager(version_file or config.get('version_file', 'extractor_version.txt'))

    def extract_folder(self, folder_path, options, progress_callback=None, log_callback=None):
        folder_path = Path(folder_path)
        default_options = config.get('extraction', {})
        merged_options = {**default_options, **options}

        extractor = CodeExtractor(
            include_json=merged_options.get('include_json', True),
            include_subdirs=merged_options.get('include_subdirs', True),
            show_file_paths=merged_options.get('show_file_paths', True),
            include_structure=merged_options.get('include_structure', True),
            include_txt=merged_options.get('include_txt', False),
            include_po=merged_options.get('include_po', False),
            include_mo=merged_options.get('include_mo', False),
            ignore_init=merged_options.get('ignore_init', False),
            include_statistics=merged_options.get('include_statistics', True),
            include_file_metadata=merged_options.get('include_file_metadata', False)
        )

        folder_name = folder_path.name
        self.output_dir.mkdir(parents=True, exist_ok=True)
        next_version = self.version_manager.get_next_version(folder_name, output_dir=self.output_dir)
        output_filename = self.output_dir / f"{folder_name}v{next_version}.txt"

        # Récupération des 6 valeurs
        success, py_count, json_count, txt_count, po_count, mo_count = extractor.extract_all(
            folder_path,
            output_filename,
            progress_callback=progress_callback,
            log_callback=log_callback
        )

        if success:
            self.version_manager.use_version(folder_name, next_version)
            stats = {
                'py_count': py_count,
                'json_count': json_count,
                'txt_count': txt_count,
                'po_count': po_count,
                'mo_count': mo_count,
                'total_files': py_count + json_count + txt_count + po_count + mo_count,
                'output_filename': str(output_filename)
            }
            logger.info(f"Extraction réussie : {output_filename}")
            return True, str(output_filename), stats
        else:
            logger.error("Échec de l'extraction")
            return False, None, None

    def clean_versions(self, archive=False):
        if archive:
            archive_dir = self.output_dir / config.get('archive_subdir', 'old_out')
            archive_dir.mkdir(parents=True, exist_ok=True)
            moved_files = []
            for item in self.output_dir.iterdir():
                if item.is_file() and item.suffix == '.txt':
                    try:
                        shutil.move(str(item), str(archive_dir / item.name))
                        moved_files.append(item.name)
                    except Exception as e:
                        logger.error(f"Erreur archivage {item} : {e}")
                        raise
            logger.info(f"Archivés : {', '.join(moved_files)}")

        self.version_manager.reset()
        logger.info("Historique des versions réinitialisé.")
        return "Nettoyage terminé" + (" (fichiers archivés)" if archive else "")
