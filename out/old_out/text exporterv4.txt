# extractor.py
import os
import json

class CodeExtractor:
    def __init__(self, include_json=True, include_subdirs=True, show_file_paths=True,
                 include_structure=True, include_txt=False):
        self.include_json = include_json
        self.include_subdirs = include_subdirs
        self.show_file_paths = show_file_paths
        self.include_structure = include_structure
        self.include_txt = include_txt

    def find_files(self, folder):
        """Trouve tous les fichiers .py, .json et éventuellement .txt dans le dossier"""
        files = []
        extensions = ['.py']
        if self.include_json:
            extensions.append('.json')
        if self.include_txt:
            extensions.append('.txt')

        if self.include_subdirs:
            for root, dirs, files_in_dir in os.walk(folder):
                for file in files_in_dir:
                    if any(file.endswith(ext) for ext in extensions):
                        full_path = os.path.join(root, file)
                        rel_path = os.path.relpath(full_path, folder)
                        ext = os.path.splitext(file)[1]
                        files.append((full_path, rel_path, ext))
        else:
            for file in os.listdir(folder):
                if any(file.endswith(ext) for ext in extensions):
                    full_path = os.path.join(folder, file)
                    ext = os.path.splitext(file)[1]
                    files.append((full_path, file, ext))
        return files

    def generate_project_structure(self, folder):
        """Génère la structure arborescente du projet"""
        structure = []
        structure.append("STRUCTURE DU PROJET")
        structure.append("=" * 80)

        root_name = os.path.basename(folder)
        structure.append(f"📁 {root_name}/")

        if self.include_subdirs:
            for root, dirs, files in os.walk(folder):
                level = root.replace(folder, '').count(os.sep)
                indent = "│   " * level
                if root != folder:
                    folder_name = os.path.basename(root)
                    structure.append(f"{indent}├── 📁 {folder_name}/")

                file_indent = "│   " * (level + 1) if root != folder else indent
                py_files = [f for f in files if f.endswith('.py')]
                json_files = [f for f in files if f.endswith('.json')] if self.include_json else []
                txt_files = [f for f in files if f.endswith('.txt')] if self.include_txt else []
                all_files = py_files + json_files + txt_files

                for i, file in enumerate(all_files):
                    is_last = (i == len(all_files) - 1)
                    prefix = "└── " if is_last else "├── "
                    if file in py_files:
                        icon = "🐍"
                    elif file in json_files:
                        icon = "📄"
                    else:  # txt
                        icon = "📝"
                    structure.append(f"{file_indent}{prefix} {icon} {file}")
        else:
            files = os.listdir(folder)
            py_files = [f for f in files if f.endswith('.py')]
            json_files = [f for f in files if f.endswith('.json')] if self.include_json else []
            txt_files = [f for f in files if f.endswith('.txt')] if self.include_txt else []
            all_files = py_files + json_files + txt_files

            for i, file in enumerate(all_files):
                is_last = (i == len(all_files) - 1)
                prefix = "└── " if is_last else "├── "
                if file in py_files:
                    icon = "🐍"
                elif file in json_files:
                    icon = "📄"
                else:
                    icon = "📝"
                structure.append(f"    {prefix} {icon} {file}")

        structure.append("\n" + "=" * 80 + "\n")
        return "\n".join(structure)

    def format_json_content(self, content, file_path):
        """Formate joliment le contenu JSON"""
        try:
            json_data = json.loads(content)
            return json.dumps(json_data, indent=2, ensure_ascii=False)
        except json.JSONDecodeError as e:
            return f"// ERREUR: JSON invalide - {str(e)}\n{content}"

    def extract_all(self, folder, output_filename, progress_callback=None, log_callback=None):
        """
        Extrait tous les fichiers vers output_filename.
        progress_callback: fonction appelée avec (current, total)
        log_callback: fonction appelée avec un message (str)
        """
        files = self.find_files(folder)
        if not files:
            if log_callback:
                log_callback("Aucun fichier trouvé")
            return False, 0, 0, 0

        total_files = len(files)
        py_count = sum(1 for _, _, ext in files if ext == '.py')
        json_count = sum(1 for _, _, ext in files if ext == '.json')
        txt_count = sum(1 for _, _, ext in files if ext == '.txt')

        with open(output_filename, 'w', encoding='utf-8') as output_file:
            # En-tête
            from src.utils import get_current_date
            output_file.write(f"Extraction du code du dossier : {folder}\n")
            output_file.write(f"Date d'extraction : {get_current_date()}\n")
            output_file.write("=" * 80 + "\n\n")

            if self.include_structure:
                structure = self.generate_project_structure(folder)
                output_file.write(structure)
                if log_callback:
                    log_callback("✓ Structure du projet générée")

            for i, (full_path, rel_path, ext) in enumerate(files):
                if progress_callback:
                    progress_callback(i + 1, total_files)

                file_type = "Python" if ext == '.py' else ("JSON" if ext == '.json' else "Texte")
                output_file.write(f"\n{'=' * 80}\n")
                output_file.write(f"FICHIER {file_type} : ")
                if self.show_file_paths:
                    output_file.write(f"{rel_path}\n")
                else:
                    output_file.write(f"{os.path.basename(full_path)}\n")
                output_file.write(f"Type: {file_type}\n")
                output_file.write(f"Chemin complet: {full_path}\n")
                output_file.write("-" * 80 + "\n")

                try:
                    with open(full_path, 'r', encoding='utf-8') as file_obj:
                        content = file_obj.read()
                        if ext == '.json':
                            content = self.format_json_content(content, full_path)
                        output_file.write(content)
                        output_file.write("\n")
                    if log_callback:
                        log_callback(f"✓ {rel_path} extrait")
                except Exception as e:
                    error_msg = f"Erreur lors de la lecture de {rel_path}: {str(e)}\n"
                    output_file.write(f"// ERREUR: {error_msg}")
                    if log_callback:
                        log_callback(f"✗ Erreur sur {rel_path}: {str(e)}")

        return True, py_count, json_count, txt_count

# versioning.py
import os

class VersionManager:
    def __init__(self, version_file="extractor_version.txt"):
        self.version_file = version_file
        self.mapping = self.load_mapping()

    def load_mapping(self):
        mapping = {}
        if os.path.exists(self.version_file):
            try:
                with open(self.version_file, "r", encoding="utf-8") as f:
                    for line in f:
                        line = line.strip()
                        if line and ":" in line:
                            folder, ver = line.split(":", 1)
                            mapping[folder] = int(ver)
            except:
                mapping = {}
        return mapping

    def save_mapping(self):
        try:
            with open(self.version_file, "w", encoding="utf-8") as f:
                for folder, ver in sorted(self.mapping.items()):
                    f.write(f"{folder}:{ver}\n")
        except:
            pass

    def get_next_version(self, folder_name, output_dir="."):
        last_version = self.mapping.get(folder_name, 0)
        version = last_version + 1
        while os.path.exists(os.path.join(output_dir, f"{folder_name}v{version}.txt")):
            version += 1
        return version

    def use_version(self, folder_name, version):
        self.mapping[folder_name] = version
        self.save_mapping()

    def reset(self):
        """Réinitialise complètement l'historique des versions."""
        self.mapping = {}
        # On efface le fichier de version
        if os.path.exists(self.version_file):
            try:
                os.remove(self.version_file)
            except:
                pass

    # Anciennes méthodes, conservées pour compatibilité
    def load_version(self):
        return 1

    def save_version(self, version):
        pass

# src/gui/extraction_controller.py
import os
import shutil
import tkinter as tk
from tkinter import filedialog, messagebox
from src.extractor import CodeExtractor
from src.versioning import VersionManager

class ExtractionController:
    def __init__(self, root, ui_widgets, version_manager):
        self.root = root
        self.ui = ui_widgets
        self.version_manager = version_manager
        self.selected_folder = ""

    def browse_folder(self):
        folder = filedialog.askdirectory(
            title="Sélectionner un dossier contenant des fichiers Python/JSON/TXT"
        )
        if not folder:
            return

        self.selected_folder = folder
        self.ui['folder_path_var'].set(folder)
        self.ui['extract_btn'].config(state='normal')
        self.clear_info()
        self.add_info(f"Dossier sélectionné : {folder}")

        temp_extractor = CodeExtractor(
            include_json=self.ui['include_json'].get(),
            include_subdirs=self.ui['include_subdirs'].get(),
            include_txt=self.ui['include_txt'].get()
        )
        files = temp_extractor.find_files(folder)
        self.add_info(f"Fichiers trouvés : {len(files)}")
        py_count = sum(1 for _, _, ext in files if ext == '.py')
        json_count = sum(1 for _, _, ext in files if ext == '.json')
        txt_count = sum(1 for _, _, ext in files if ext == '.txt')
        self.add_info(f"  - Fichiers Python : {py_count}")
        self.add_info(f"  - Fichiers JSON : {json_count}")
        self.add_info(f"  - Fichiers TXT : {txt_count}")

    def extract_code(self):
        if not self.selected_folder:
            messagebox.showwarning("Attention", "Veuillez d'abord sélectionner un dossier")
            return

        extractor = CodeExtractor(
            include_json=self.ui['include_json'].get(),
            include_subdirs=self.ui['include_subdirs'].get(),
            show_file_paths=self.ui['show_file_paths'].get(),
            include_structure=self.ui['include_structure'].get(),
            include_txt=self.ui['include_txt'].get()
        )

        folder_name = os.path.basename(self.selected_folder)
        output_dir = "out"
        os.makedirs(output_dir, exist_ok=True)

        next_version = self.version_manager.get_next_version(folder_name, output_dir=output_dir)
        output_filename = os.path.join(output_dir, f"{folder_name}v{next_version}.txt")

        self.add_info(f"\n--- Extraction en cours vers {output_filename} ---")
        self.ui['status_var'].set("Extraction en cours...")

        def update_progress(current, total):
            progress = (current / total) * 100
            self.ui['progress_var'].set(progress)
            self.root.update_idletasks()

        def log_message(msg):
            self.add_info(msg)

        success, py_count, json_count, txt_count = extractor.extract_all(
            self.selected_folder,
            output_filename,
            progress_callback=update_progress,
            log_callback=log_message
        )

        if not success:
            self.ui['status_var'].set("Extraction échouée")
            self.ui['progress_var'].set(0)
            messagebox.showerror("Erreur", "Aucun fichier trouvé ou erreur lors de l'extraction")
            return

        self.version_manager.use_version(folder_name, next_version)

        self.ui['status_var'].set("Extraction terminée")
        self.ui['progress_var'].set(0)

        total_files = py_count + json_count + txt_count
        success_msg = (
            f"Extraction terminée avec succès !\n\n"
            f"Fichier créé : {output_filename}\n"
            f"Nombre de fichiers extraits : {total_files}\n"
            f"  - Fichiers Python : {py_count}\n"
            f"  - Fichiers JSON : {json_count}\n"
            f"  - Fichiers TXT : {txt_count}\n"
            f"Emplacement : {os.path.abspath(output_filename)}"
        )
        self.add_info(f"\n--- Extraction terminée ---")
        self.add_info(success_msg)
        messagebox.showinfo("Succès", success_msg)

    def clean_versions(self):
        """Nettoie l'historique des versions (appelé depuis le menu Maintenance)."""
        archive = self.ui['archive_old'].get()

        if not messagebox.askyesno("Confirmation",
                                   "Réinitialiser toutes les versions et l'historique ?\n"
                                   + ("Les anciens fichiers seront archivés dans out/old_out/." if archive else "")):
            return

        output_dir = "out"

        if archive:
            archive_dir = os.path.join(output_dir, "old_out")
            os.makedirs(archive_dir, exist_ok=True)

            for item in os.listdir(output_dir):
                item_path = os.path.join(output_dir, item)
                if os.path.isfile(item_path) and item.endswith(".txt"):
                    try:
                        shutil.move(item_path, os.path.join(archive_dir, item))
                        self.add_info(f"Archivé : {item} -> out/old_out/")
                    except Exception as e:
                        self.add_info(f"Erreur archivage {item} : {e}")

        self.version_manager.reset()
        self.add_info("Historique des versions réinitialisé.")
        self.ui['status_var'].set("Nettoyage terminé")

    def add_info(self, message):
        self.ui['info_text'].insert(tk.END, message + "\n")
        self.ui['info_text'].see(tk.END)
        self.root.update_idletasks()

    def clear_info(self):
        self.ui['info_text'].delete(1.0, tk.END)
