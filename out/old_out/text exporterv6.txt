# versioning.py
import os

class VersionManager:
    def __init__(self, version_file="extractor_version.txt"):
        self.version_file = version_file
        self.mapping = self.load_mapping()

    def load_mapping(self):
        mapping = {}
        if os.path.exists(self.version_file):
            try:
                with open(self.version_file, "r", encoding="utf-8") as f:
                    for line in f:
                        line = line.strip()
                        if line and ":" in line:
                            folder, ver = line.split(":", 1)
                            mapping[folder] = int(ver)
            except:
                mapping = {}
        return mapping

    def save_mapping(self):
        try:
            with open(self.version_file, "w", encoding="utf-8") as f:
                for folder, ver in sorted(self.mapping.items()):
                    f.write(f"{folder}:{ver}\n")
        except:
            pass

    def get_next_version(self, folder_name, output_dir="."):
        last_version = self.mapping.get(folder_name, 0)
        version = last_version + 1
        while os.path.exists(os.path.join(output_dir, f"{folder_name}v{version}.txt")):
            version += 1
        return version

    def use_version(self, folder_name, version):
        self.mapping[folder_name] = version
        self.save_mapping()

    def reset(self):
        """Réinitialise complètement l'historique des versions."""
        self.mapping = {}
        # On efface le fichier de version
        if os.path.exists(self.version_file):
            try:
                os.remove(self.version_file)
            except:
                pass

    # Anciennes méthodes, conservées pour compatibilité
    def load_version(self):
        return 1

    def save_version(self, version):
        pass

# src/extractor/content_formatter.py
import json

def format_json_content(content, file_path=None):
    """Essaie de formater joliment le contenu JSON.
       En cas d'échec, retourne le contenu brut précédé d'un commentaire d'erreur.
    """
    try:
        json_data = json.loads(content)
        return json.dumps(json_data, indent=2, ensure_ascii=False)
    except json.JSONDecodeError as e:
        return f"// ERREUR: JSON invalide - {str(e)}\n{content}"

# src/extractor/code_extractor.py
import os
from .file_finder import find_files
from .structure_generator import generate_project_structure
from .content_formatter import format_json_content
from .stats_writer import write_statistics_section

class CodeExtractor:
    def __init__(self, include_json=True, include_subdirs=True, show_file_paths=True,
                 include_structure=True, include_txt=False, ignore_init=False,
                 include_statistics=False):
        self.include_json = include_json
        self.include_subdirs = include_subdirs
        self.show_file_paths = show_file_paths
        self.include_structure = include_structure
        self.include_txt = include_txt
        self.ignore_init = ignore_init
        self.include_statistics = include_statistics

    def find_files(self, folder):
        return find_files(folder,
                          include_json=self.include_json,
                          include_subdirs=self.include_subdirs,
                          include_txt=self.include_txt,
                          ignore_init=self.ignore_init)

    def generate_project_structure(self, folder):
        return generate_project_structure(folder,
                                          include_json=self.include_json,
                                          include_subdirs=self.include_subdirs,
                                          include_txt=self.include_txt,
                                          ignore_init=self.ignore_init)

    def format_json_content(self, content, file_path=None):
        return format_json_content(content, file_path)

    def extract_all(self, folder, output_filename, progress_callback=None, log_callback=None):
        """
        Extrait tous les fichiers vers output_filename.
        progress_callback: fonction appelée avec (current, total)
        log_callback: fonction appelée avec un message (str)
        Retourne (success, py_count, json_count, txt_count)
        """
        files = self.find_files(folder)
        if not files:
            if log_callback:
                log_callback("Aucun fichier trouvé")
            return False, 0, 0, 0

        total_files = len(files)
        py_count = sum(1 for _, _, ext in files if ext == '.py')
        json_count = sum(1 for _, _, ext in files if ext == '.json')
        txt_count = sum(1 for _, _, ext in files if ext == '.txt')

        total_lines_py = 0
        total_lines_json = 0
        total_lines_txt = 0
        total_size = 0

        with open(output_filename, 'w', encoding='utf-8') as output_file:
            # En-tête
            from src.utils import get_current_date
            output_file.write(f"Extraction du code du dossier : {folder}\n")
            output_file.write(f"Date d'extraction : {get_current_date()}\n")
            output_file.write("=" * 80 + "\n\n")

            if self.include_structure:
                structure = self.generate_project_structure(folder)
                output_file.write(structure)
                if log_callback:
                    log_callback("✓ Structure du projet générée")

            for i, (full_path, rel_path, ext) in enumerate(files):
                if progress_callback:
                    progress_callback(i + 1, total_files)

                file_type = "Python" if ext == '.py' else ("JSON" if ext == '.json' else "Texte")
                output_file.write(f"\n{'=' * 80}\n")
                output_file.write(f"FICHIER {file_type} : ")
                if self.show_file_paths:
                    output_file.write(f"{rel_path}\n")
                else:
                    output_file.write(f"{os.path.basename(full_path)}\n")
                output_file.write(f"Type: {file_type}\n")
                output_file.write(f"Chemin complet: {full_path}\n")
                output_file.write("-" * 80 + "\n")

                try:
                    with open(full_path, 'r', encoding='utf-8') as file_obj:
                        content = file_obj.read()
                        if ext == '.json':
                            content = self.format_json_content(content, full_path)
                        num_lines = len(content.splitlines())
                        file_size = os.path.getsize(full_path)
                        total_size += file_size

                        if ext == '.py':
                            total_lines_py += num_lines
                        elif ext == '.json':
                            total_lines_json += num_lines
                        else:
                            total_lines_txt += num_lines

                        output_file.write(content)
                        output_file.write("\n")
                    if log_callback:
                        log_callback(f"✓ {rel_path} extrait")
                except Exception as e:
                    error_msg = f"Erreur lors de la lecture de {rel_path}: {str(e)}\n"
                    output_file.write(f"// ERREUR: {error_msg}")
                    if log_callback:
                        log_callback(f"✗ Erreur sur {rel_path}: {str(e)}")

            if self.include_statistics:
                write_statistics_section(output_file, folder, self.include_subdirs,
                                         py_count, json_count, txt_count,
                                         total_lines_py, total_lines_json, total_lines_txt,
                                         total_size)

        return True, py_count, json_count, txt_count

# src/extractor/file_finder.py
import os

def find_files(folder, include_json=True, include_subdirs=True,
               include_txt=False, ignore_init=False):
    """Trouve tous les fichiers .py, .json et éventuellement .txt dans le dossier.
       Retourne une liste de tuples (chemin_complet, chemin_relatif, extension).
    """
    files = []
    extensions = ['.py']
    if include_json:
        extensions.append('.json')
    if include_txt:
        extensions.append('.txt')

    if include_subdirs:
        for root, dirs, files_in_dir in os.walk(folder):
            for file in files_in_dir:
                if any(file.endswith(ext) for ext in extensions):
                    if ignore_init and file == '__init__.py':
                        continue
                    full_path = os.path.join(root, file)
                    rel_path = os.path.relpath(full_path, folder)
                    ext = os.path.splitext(file)[1]
                    files.append((full_path, rel_path, ext))
    else:
        for file in os.listdir(folder):
            if any(file.endswith(ext) for ext in extensions):
                if ignore_init and file == '__init__.py':
                    continue
                full_path = os.path.join(folder, file)
                ext = os.path.splitext(file)[1]
                files.append((full_path, file, ext))
    return files

# src/extractor/stats_writer.py
import os

def write_statistics_section(output_file, folder, include_subdirs,
                             py_count, json_count, txt_count,
                             total_lines_py, total_lines_json, total_lines_txt,
                             total_size):
    """Écrit la section STATISTIQUES dans le fichier ouvert."""
    # Calcul des dossiers et packages
    num_dirs = 0
    num_packages = 0
    if include_subdirs:
        for root, dirs, f_in_dir in os.walk(folder):
            if root != folder:
                num_dirs += 1
            if '__init__.py' in f_in_dir:
                num_packages += 1
    else:
        if '__init__.py' in os.listdir(folder):
            num_packages = 1
        num_dirs = 0

    output_file.write("\n" + "=" * 80 + "\n")
    output_file.write("STATISTIQUES\n")
    output_file.write("=" * 80 + "\n")
    output_file.write(f"Nombre de dossiers          : {num_dirs}\n")
    output_file.write(f"Dossiers 'Python package'    : {num_packages}\n")
    output_file.write(f"Fichiers Python (.py)        : {py_count}\n")
    output_file.write(f"Fichiers JSON (.json)        : {json_count}\n")
    output_file.write(f"Fichiers Texte (.txt)        : {txt_count}\n")
    output_file.write(f"Nombre total de fichiers     : {py_count + json_count + txt_count}\n")
    output_file.write(f"Nombre de lignes (Python)    : {total_lines_py}\n")
    output_file.write(f"Nombre de lignes (JSON)      : {total_lines_json}\n")
    output_file.write(f"Nombre de lignes (Texte)     : {total_lines_txt}\n")
    output_file.write(f"Nombre total de lignes       : {total_lines_py + total_lines_json + total_lines_txt}\n")

    # Taille lisible
    if total_size < 1024:
        size_str = f"{total_size} octets"
    elif total_size < 1024 * 1024:
        size_str = f"{total_size / 1024:.2f} Ko"
    else:
        size_str = f"{total_size / (1024 * 1024):.2f} Mo"
    output_file.write(f"Volume total extrait         : {size_str}\n")

# src/extractor/structure_generator.py
import os

def generate_project_structure(folder, include_json=True, include_subdirs=True,
                               include_txt=False, ignore_init=False):
    """Génère une représentation textuelle de l'arborescence du projet."""
    lines = []
    lines.append("STRUCTURE DU PROJET")
    lines.append("=" * 80)

    root_name = os.path.basename(folder)
    lines.append(f"📁 {root_name}/")

    if include_subdirs:
        for root, dirs, files in os.walk(folder):
            level = root.replace(folder, '').count(os.sep)
            indent = "│   " * level
            if root != folder:
                folder_name = os.path.basename(root)
                lines.append(f"{indent}├── 📁 {folder_name}/")

            file_indent = "│   " * (level + 1) if root != folder else indent
            py_files = [f for f in files if f.endswith('.py')]
            if ignore_init:
                py_files = [f for f in py_files if f != '__init__.py']
            json_files = [f for f in files if f.endswith('.json')] if include_json else []
            txt_files = [f for f in files if f.endswith('.txt')] if include_txt else []
            all_files = py_files + json_files + txt_files

            for i, file in enumerate(all_files):
                is_last = (i == len(all_files) - 1)
                prefix = "└── " if is_last else "├── "
                if file in py_files:
                    icon = "🐍"
                elif file in json_files:
                    icon = "📄"
                else:  # txt
                    icon = "📝"
                lines.append(f"{file_indent}{prefix} {icon} {file}")
    else:
        files = os.listdir(folder)
        py_files = [f for f in files if f.endswith('.py')]
        if ignore_init:
            py_files = [f for f in py_files if f != '__init__.py']
        json_files = [f for f in files if f.endswith('.json')] if include_json else []
        txt_files = [f for f in files if f.endswith('.txt')] if include_txt else []
        all_files = py_files + json_files + txt_files

        for i, file in enumerate(all_files):
            is_last = (i == len(all_files) - 1)
            prefix = "└── " if is_last else "├── "
            if file in py_files:
                icon = "🐍"
            elif file in json_files:
                icon = "📄"
            else:
                icon = "📝"
            lines.append(f"    {prefix} {icon} {file}")

    lines.append("\n" + "=" * 80 + "\n")
    return "\n".join(lines)

import os

import tkinter as tk
from tkinter import filedialog, messagebox

class ExtractionController:
    def __init__(self, root, ui_widgets, version_manager):
        self.root = root
        self.ui = ui_widgets
        self.version_manager = version_manager
        self.selected_folder = ""

    def browse_folder(self):
        folder = filedialog.askdirectory(

        )
        if not folder:
            return

        self.selected_folder = folder
        self.ui['folder_path_var'].set(folder)
        self.ui['extract_btn'].config(state='normal')
        self.clear_info()
        self.add_info(f"Dossier sélectionné : {folder}")

        temp_extractor = CodeExtractor(
            include_json=self.ui['include_json'].get(),
            include_subdirs=self.ui['include_subdirs'].get(),
            include_txt=self.ui['include_txt'].get(),
            ignore_init=self.ui['ignore_init'].get()           # Ajout
        )
        files = temp_extractor.find_files(folder)
        self.add_info(f"Fichiers trouvés : {len(files)}")
        py_count = sum(1 for _, _, ext in files if ext == '.py')
        json_count = sum(1 for _, _, ext in files if ext == '.json')
        txt_count = sum(1 for _, _, ext in files if ext == '.txt')
        self.add_info(f"  - Fichiers Python : {py_count}")
        self.add_info(f"  - Fichiers JSON : {json_count}")
        self.add_info(f"  - Fichiers TXT : {txt_count}")

    def extract_code(self):
        if not self.selected_folder:
            messagebox.showwarning("Attention", "Veuillez d'abord sélectionner un dossier")
            return

        extractor = CodeExtractor(
            include_json=self.ui['include_json'].get(),
            include_subdirs=self.ui['include_subdirs'].get(),
            show_file_paths=self.ui['show_file_paths'].get(),
            include_structure=self.ui['include_structure'].get(),
            include_txt=self.ui['include_txt'].get(),
            ignore_init=self.ui['ignore_init'].get(),
            include_statistics=self.ui['include_statistics'].get()  # Nouveau
        )

        folder_name = os.path.basename(self.selected_folder)
        output_dir = "out"
        os.makedirs(output_dir, exist_ok=True)

        next_version = self.version_manager.get_next_version(folder_name, output_dir=output_dir)
        output_filename = os.path.join(output_dir, f"{folder_name}v{next_version}.txt")

        self.add_info(f"\n--- Extraction en cours vers {output_filename} ---")
        self.ui['status_var'].set("Extraction en cours...")

        def update_progress(current, total):
            progress = (current / total) * 100
            self.ui['progress_var'].set(progress)
            self.root.update_idletasks()

        def log_message(msg):
            self.add_info(msg)

        success, py_count, json_count, txt_count = extractor.extract_all(
            self.selected_folder,
            output_filename,
            progress_callback=update_progress,
            log_callback=log_message
        )

        if not success:
            self.ui['status_var'].set("Extraction échouée")
            self.ui['progress_var'].set(0)
            messagebox.showerror("Erreur", "Aucun fichier trouvé ou erreur lors de l'extraction")
            return

        self.version_manager.use_version(folder_name, next_version)

        self.ui['status_var'].set("Extraction terminée")
        self.ui['progress_var'].set(0)

        total_files = py_count + json_count + txt_count
        success_msg = (
            f"Extraction terminée avec succès !\n\n"
            f"Fichier créé : {output_filename}\n"
            f"Nombre de fichiers extraits : {total_files}\n"
            f"  - Fichiers Python : {py_count}\n"
            f"  - Fichiers JSON : {json_count}\n"
            f"  - Fichiers TXT : {txt_count}\n"
            f"Emplacement : {os.path.abspath(output_filename)}"
        )
        self.add_info(f"\n--- Extraction terminée ---")
        self.add_info(success_msg)
        messagebox.showinfo("Succès", success_msg)

    def clean_versions(self):
        """Nettoie l'historique des versions (appelé depuis le menu Maintenance)."""
        archive = self.ui['archive_old'].get()

        if not messagebox.askyesno("Confirmation",
                                   "Réinitialiser toutes les versions et l'historique ?\n"
                                   + ("Les anciens fichiers seront archivés dans out/old_out/." if archive else "")):
            return

        output_dir = "out"

        if archive:
            archive_dir = os.path.join(output_dir, "old_out")
            os.makedirs(archive_dir, exist_ok=True)

            for item in os.listdir(output_dir):
                item_path = os.path.join(output_dir, item)
                if os.path.isfile(item_path) and item.endswith(".txt"):
                    try:
                        shutil.move(item_path, os.path.join(archive_dir, item))
                        self.add_info(f"Archivé : {item} -> out/old_out/")
                    except Exception as e:
                        self.add_info(f"Erreur archivage {item} : {e}")

        self.version_manager.reset()
        self.add_info("Historique des versions réinitialisé.")
        self.ui['status_var'].set("Nettoyage terminé")

    def add_info(self, message):
        self.ui['info_text'].insert(tk.END, message + "\n")
        self.ui['info_text'].see(tk.END)
        self.root.update_idletasks()

    def clear_info(self):
        self.ui['info_text'].delete(1.0, tk.END)

# src/gui/ui_builder/menus.py
import tkinter as tk

def build_menus(parent, ui):
    """Crée la barre de menus et les sous-menus.
       Attend un dict `ui` déjà existant (muté).
    """
    menubar = tk.Menu(parent)
    parent.config(menu=menubar)
    ui['menubar'] = menubar

    # Menu Fichier
    file_menu = tk.Menu(menubar, tearoff=0)
    menubar.add_cascade(label="Fichier", menu=file_menu)
    file_menu.add_command(label="Quitter", command=parent.destroy, accelerator="Ctrl+Q")
    parent.bind_all("<Control-q>", lambda e: parent.destroy())

    # Menu Options

# src/gui/ui_builder/widgets.py
import tkinter as tk
from tkinter import ttk

def build_widgets(parent, ui):
    # --- Interface principale ---
    main_frame = ttk.Frame(parent, padding="10")
    main_frame.grid(row=0, column=0, sticky=(tk.W, tk.E, tk.N, tk.S))
    ui['main_frame'] = main_frame

    # Titre
    title_label = ttk.Label(main_frame, text="Extracteur de code Python, JSON & TXT",
                            font=('Arial', 16, 'bold'))
    title_label.grid(row=0, column=0, columnspan=2, pady=10)

    # Sélection du dossier
    folder_label = ttk.Label(main_frame, text="Dossier à scanner :")
    folder_label.grid(row=1, column=0, sticky=tk.W, pady=5)

    folder_entry = ttk.Entry(main_frame, textvariable=ui['folder_path_var'],
                             width=60, state='readonly')
    folder_entry.grid(row=1, column=1, padx=5, pady=5, sticky=tk.W)

    browse_btn = ttk.Button(main_frame, text="Parcourir")
    browse_btn.grid(row=1, column=2, padx=5, pady=5)
    ui['browse_btn'] = browse_btn

    # Boutons d'action
    button_frame = ttk.Frame(main_frame)
    button_frame.grid(row=2, column=0, columnspan=3, pady=10)

    extract_btn = ttk.Button(button_frame, text="Extraire le code", state='disabled')
    extract_btn.pack(side=tk.LEFT, padx=5)
    ui['extract_btn'] = extract_btn

    clear_btn = ttk.Button(button_frame, text="Effacer le journal")
    clear_btn.pack(side=tk.LEFT, padx=5)
    ui['clear_btn'] = clear_btn

    # Barre de progression
    progress_bar = ttk.Progressbar(main_frame, variable=ui['progress_var'],
                                   maximum=100, length=500)
    progress_bar.grid(row=3, column=0, columnspan=3, pady=10)
    ui['progress_bar'] = progress_bar

    # Zone d'informations (journal) – masquable
    info_frame = ttk.LabelFrame(main_frame, text="Journal", padding="5")
    info_frame.grid(row=4, column=0, columnspan=3, sticky=(tk.W, tk.E, tk.N, tk.S), pady=5)
    ui['info_frame'] = info_frame

    info_text = tk.Text(info_frame, height=12, width=80, wrap=tk.WORD)
    info_text.grid(row=0, column=0, sticky=(tk.W, tk.E, tk.N, tk.S))

    scrollbar = ttk.Scrollbar(info_frame, orient="vertical", command=info_text.yview)
    scrollbar.grid(row=0, column=1, sticky=(tk.N, tk.S))
    info_text.configure(yscrollcommand=scrollbar.set)
    ui['info_text'] = info_text

    # Status bar
    status_bar = ttk.Label(main_frame, textvariable=ui['status_var'],
                           relief=tk.SUNKEN, anchor=tk.W)
    status_bar.grid(row=5, column=0, columnspan=3, sticky=(tk.W, tk.E), pady=5)
    ui['status_bar'] = status_bar

    # Poids des colonnes/lignes
    main_frame.columnconfigure(1, weight=1)
    main_frame.rowconfigure(4, weight=1)
    info_frame.columnconfigure(0, weight=1)
    info_frame.rowconfigure(0, weight=1)
    parent.columnconfigure(0, weight=1)
    parent.rowconfigure(0, weight=1)

    # --- Fonction de bascule du journal ---
    def toggle_log_visibility():
        if ui['log_visible'].get():
            ui['info_frame'].grid()
        else:
            ui['info_frame'].grid_remove()

    view_menu = ui.get('view_menu')
    if view_menu:
        view_menu.entryconfig(0, command=toggle_log_visibility)
